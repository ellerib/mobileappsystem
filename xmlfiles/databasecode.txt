CREATE TABLE users(
    user_id INT(10) PRIMARY KEY,
    username VARCHAR(20),
    email VARCHAR(100),
    role VARCHAR(50),
    password_hash VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE purchases(
    purchase_id VARCHAR(50) PRIMARY KEY,
    customer_name VARCHAR(50),
    purchase_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_activated BOOLEAN DEFAULT TRUE
);

CREATE TABLE tank(
    tank_id INT(11) PRIMARY KEY,
    purchase_id VARCHAR(50),
    tankname VARCHAR(100),

    FOREIGN KEY (purchase_id) REFERENCES purchases(purchase_id)
);

CREATE TABLE alerts(
    alert_id INT(11) PRIMARY KEY,
    tank_id INT(11),
    alert_type VARCHAR(50),
    message TEXT,
    status ENUM('Read','Unread'),
    alert_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (tank_id) REFERENCES tank(tank_id)
);

CREATE TABLE dashboard(
	dashboard_id INT(10) NOT NULL PRIMARY KEY,
    mode ENUM('Growing', 'Breeding'),
    temperature Decimal(5, 2),
    turbidity Decimal(5, 2),
    ph_Level Decimal(5, 2),
    status VARCHAR(50),
    user_id INT,
    tank_id INT,
 
    FOREIGN KEY (user_id) REFERENCES users(user_id),
    FOREIGN KEY (tank_id) REFERENCES tank(tank_id)
);
